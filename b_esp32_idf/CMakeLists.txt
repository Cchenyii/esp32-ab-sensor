cmake_minimum_required(VERSION 3.22)

set(PROJECT_VER "0.7.1")
if(DEFINED B_OTA_BUILD_VERSION AND NOT "${B_OTA_BUILD_VERSION}" STREQUAL "")
    set(PROJECT_VER "${B_OTA_BUILD_VERSION}")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(esp32_b_idf)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/.idf-version" expected_idf_tag)
string(STRIP "${expected_idf_tag}" expected_idf_tag)
idf_build_get_property(actual_idf_tag IDF_VER)
if(NOT "${actual_idf_tag}" STREQUAL "${expected_idf_tag}")
    message(FATAL_ERROR "This project requires ESP-IDF ${expected_idf_tag}; found ${actual_idf_tag}")
endif()
if(NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "This board project requires IDF_TARGET=esp32")
endif()
